🔍 PENYEBAB MASALAH (SISTEM LAMA):

1. Machine ID Berubah Tanpa Alasan
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   Penyebab:
   • Menggunakan $_SERVER['SERVER_ADDR'] (bisa berubah karena DHCP)
   • Menggunakan $_SERVER['HTTP_HOST'] (bisa berubah karena domain alias)
   • Komponen tidak stabil
   
   Dampak:
   ❌ Machine ID berubah → Validasi gagal → Minta aktivasi ulang

2. Hash Mismatch Karena Data Berubah
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   Penyebab:
   • Update expires_at dari server
   • Hash tidak di-regenerate
   • Hash validation gagal
   
   Dampak:
   ❌ Hash mismatch → Block → Minta aktivasi ulang

3. Network Issue → Suspend License
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   Penyebab:
   • Server lisensi unreachable 3x
   • Langsung suspend license
   • Terlalu agresif
   
   Dampak:
   ❌ License suspended → Minta aktivasi ulang

4. File .install_id Terhapus
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   Penyebab:
   • User hapus folder writable/
   • Disk full/corrupt
   • Backup restore incomplete
   
   Dampak:
   ❌ Generate install_id baru → Machine ID berubah → Minta aktivasi ulang

5. Domain Berubah
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   Penyebab:
   • User ganti domain
   • Strict validation (exact match)
   • Tidak ada auto-update
   
   Dampak:
   ❌ Domain tidak match → Block → Minta aktivasi ulang

✅ PERBAIKAN YANG SUDAH DIIMPLEMENTASIKAN:

FIX #1: Stabilkan Machine ID Generation
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SEBELUM (Tidak Stabil):
$data[] = $_SERVER['SERVER_ADDR'];      // ❌ DHCP bisa berubah
$data[] = $_SERVER['HTTP_HOST'];        // ❌ Domain alias bisa berubah

SESUDAH (Stabil):
$data[] = $installId;                    // ✅ Persistent di file
$data[] = php_uname('n');                // ✅ Hostname stabil
$data[] = php_uname('s');                // ✅ OS type stabil
$data[] = php_uname('m');                // ✅ Architecture stabil
$data[] = get_current_user();            // ✅ System user stabil
$data[] = md5($_SERVER['DOCUMENT_ROOT']); // ✅ Document root stabil

KEUNTUNGAN:
✅ Installation ID persistent di file
✅ Komponen yang digunakan stabil
✅ Machine ID tidak berubah kecuali hardware/OS berubah

---

FIX #2: Auto-Regenerate Hash Saat Update
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SEBELUM (Manual):
$model->update($id, [
    'expires_at' => $newExpiry
    // ❌ Hash tidak di-update → mismatch!
]);

SESUDAH (Otomatis):
$updateData = [
    'expires_at' => $newExpiry
];
// ✅ AUTO-REGENERATE HASH
$updateData['hash'] = hash('sha256', $key . $newExpiry . $machineId . $secret);
$model->update($id, $updateData);

KEUNTUNGAN:
✅ Hash selalu sinkron dengan data
✅ Tidak ada hash mismatch
✅ Update dari server tidak menyebabkan masalah

---

FIX #3: Graceful Handling untuk Network Issue
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SEBELUM (Terlalu Agresif):
if ($failCount > 3) {
    $model->update($id, ['status' => 'suspended']);  // ❌ SUSPEND!
}

SESUDAH (Lebih Toleran):
catch (\Exception $e) {
    log_message('error', 'Verification failed');
    $failCount++;
    
    if ($failCount > 3) {
        $model->update($id, ['status' => 'suspended']);
    }
    
    // ✅ TIDAK BLOCK aplikasi, hanya log error
    // Aplikasi tetap jalan dengan lisensi lokal
}

KEUNTUNGAN:
✅ Network issue sementara tidak menyebabkan suspend
✅ Aplikasi tetap berjalan dengan lisensi lokal
✅ Grace period 3x (72 jam)

---

FIX #4: Protect Installation ID File
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SEBELUM (Rentan Hilang):
$installFile = WRITEPATH . '.install_id';
// File bisa terhapus tanpa proteksi

SESUDAH (Protected + Backup):
// 1. Try from .env first (backup)
$installId = env('license.installId');

// 2. Fallback to file
if (!$installId) {
    $installFile = WRITEPATH . '.install_id';
    $installId = file_get_contents($installFile);
}

// 3. Generate new only if both missing
if (!$installId) {
    $installId = bin2hex(random_bytes(32));
    log_message('warning', 'Generated new installation ID');
}

KEUNTUNGAN:
✅ Cache di memory (tidak perlu baca file berulang)
✅ Backup di .env (jika file hilang)
✅ Log warning jika generate baru (untuk monitoring)

---

FIX #5: Flexible Domain Validation
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SEBELUM (Strict):
if ($license['domain'] !== $currentDomain) {
    self::lock("INVALID DOMAIN");  // ❌ BLOCK!
}

SESUDAH (Flexible + Auto-Update):
// 1. Skip untuk localhost
if (is_local_environment()) {
    return;
}

// 2. Hanya validasi jika domain sudah di-set
if (!empty($license['domain']) && $license['domain'] !== $currentDomain) {
    self::lock("INVALID DOMAIN");
}

// 3. Auto-update domain saat periodic verification
$updateData['domain'] = $domain;  // ✅ Auto-update

KEUNTUNGAN:
✅ Domain bisa berubah (dengan approval dari server)
✅ Auto-update domain saat periodic verification
✅ Tidak block jika domain belum di-set
